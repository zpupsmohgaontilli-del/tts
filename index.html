    function splitIntoClauses(text, maxLen = 170) {

      const rawParts = humaniseToggle.checked

        ? text.split(/([.!?।,;:\n]+)/).reduce((acc, cur, idx, arr) => {

            if (idx % 2 === 0) {

              const s = (cur + (arr[idx + 1] || "")).trim();

              if (s) acc.push(s);

            }

            return acc;

          }, [])

        : [text.trim()];

      const chunks = [];

      for (const part of rawParts) {

        if (part.length <= maxLen) {

          chunks.push(part);

        } else {

          const words = part.split(/\s+/);

          let line = "";

          for (const w of words) {

            if ((line + " " + w).trim().length > maxLen) {

              if (line) chunks.push(line.trim());

              line = w;

            } else {

              line = (line + " " + w).trim();

            }

          }

          if (line) chunks.push(line.trim());

        }

      }

      return chunks.length ? chunks : [text.trim()];

    }

    function getCloudTtsUrl(textChunk, langCode) {

      const q = encodeURIComponent(textChunk);

      return `https://translate.google.com/translate_tts?ie=UTF-8&client=tw-ob&tl=${encodeURIComponent(langCode)}&q=${q}`;

    }

        playSystemClauses(clauses, sysVoice, baseRate, basePitch, 0);

      } else {

        playCloudClauses(clauses, currentProfile.cloudLang, baseRate, 0);

      }

    }

    function playSystemClauses(clauses, voice, baseRate, basePitch, idx) {

      if (!isPlaying || idx >= clauses.length) {

        stopAllAudio();

        statusText.textContent = "Finished speaking.";

        return;

      }

      const chunk = clauses[idx];

      const u = new SpeechSynthesisUtterance(chunk);

      if (voice) {

        u.voice = voice;

        u.lang = voice.lang;

      } else {

        u.lang = currentProfile.lang;

      }

      if (humaniseToggle.checked) {

        const isQ = chunk.endsWith("?");

        const jitter = ((idx % 3) - 1) * 0.015;

        u.rate = Math.max(0.5, Math.min(2, baseRate + jitter));

        u.pitch = Math.max(0.5, Math.min(2, basePitch + (isQ ? 0.05 : jitter)));

      } else {

        u.rate = baseRate;

        u.pitch = basePitch;

      }

      statusText.textContent = `Speaking clause ${idx + 1}/${clauses.length}...`;

      u.onend = () => {

        if (!isPlaying) return;

        const pause = humaniseToggle.checked ? 135 : 0;

        activeTimer = setTimeout(() => playSystemClauses(clauses, voice, baseRate, basePitch, idx + 1), pause);

      };

      u.onerror = () => {

        statusText.textContent = "Switching to Cloud Neural stream...";

        playCloudClauses(clauses, currentProfile.cloudLang, baseRate, idx);

      };

      window.speechSynthesis.speak(u);

    }

    function playCloudClauses(clauses, langCode, baseRate, idx) {

      if (!isPlaying || idx >= clauses.length) {

        stopAllAudio();

        statusText.textContent = "Finished speaking.";

        return;

      }

      const chunk = clauses[idx];

      cloudPlayer.src = getCloudTtsUrl(chunk, langCode);

      cloudPlayer.playbackRate = baseRate;

      statusText.textContent = `Playing Neural Audio ${idx + 1}/${clauses.length}...`;

      cloudPlayer.onended = () => {

        if (!isPlaying) return;

        const pause = humaniseToggle.checked ? 135 : 0;

        activeTimer = setTimeout(() => playCloudClauses(clauses, langCode, baseRate, idx + 1), pause);

      };

      cloudPlayer.onerror = () => {

        if ("speechSynthesis" in window) {

          statusText.textContent = "Using device voice fallback...";

          playSystemClauses(clauses, null, baseRate, parseFloat(pitchSlider.value), idx);

        } else {

          stopAllAudio();

          statusText.textContent = "Check internet connection for Cloud MP3.";

        }

      };

      cloudPlayer.play().catch(() => {

        stopAllAudio();

        statusText.textContent = "Tap ▶ on the MP3 Stream bar below to play.";

      });

    }

    async function downloadAudio() {

      const text = scriptInput.value.trim();

      if (!text) {

        statusText.textContent = "Enter text before saving MP3.";

        return;

      }

      downloadBtn.disabled = true;

      downloadBtn.textContent = "⏳ Saving...";

      statusText.textContent = "Building MP3 file...";

      const clauses = splitIntoClauses(text, 170);

      const firstUrl = getCloudTtsUrl(clauses[0], currentProfile.cloudLang);

      cloudPlayer.src = firstUrl;

      try {

        const buffers = [];

        for (let i = 0; i < clauses.length; i++) {

          statusText.textContent = `Encoding part ${i + 1}/${clauses.length}...`;

          const url = getCloudTtsUrl(clauses[i], currentProfile.cloudLang);

          const resp = await fetch(url);

          if (!resp.ok) throw new Error("CORS fallback");

          const buf = await resp.arrayBuffer();

          buffers.push(buf);

        }

        const mergedBlob = new Blob(buffers, { type: "audio/mpeg" });

        const blobUrl = URL.createObjectURL(mergedBlob);

        cloudPlayer.src = blobUrl;

        const a = document.createElement("a");

        a.href = blobUrl;

        a.download = `${currentProfile.id}-voice.mp3`;

        document.body.appendChild(a);

        a.click();

        a.remove();

        statusText.textContent = `Saved ${currentProfile.id}-voice.mp3!`;

      } catch (err) {

        const a = document.createElement("a");

        a.href = firstUrl;

        a.target = "_blank";

        a.rel = "noopener";

        a.download = `${currentProfile.id}-voice.mp3`;

        document.body.appendChild(a);

        a.click();

        a.remove();

        statusText.textContent = "MP3 loaded in stream bar (tap ⋮ on player to download).";

      } finally {

        downloadBtn.disabled = false;

        downloadBtn.textContent = "⬇ Save MP3";

      }

    }

    rateSlider.addEventListener("input", () => {

      rateVal.textContent = parseFloat(rateSlider.value).toFixed(2) + "x";

    });

    pitchSlider.addEventListener("input", () => {

      pitchVal.textContent = parseFloat(pitchSlider.value).toFixed(2);

    });

    scriptInput.addEventListener("change", updatePlayerStreamPreview);

    voiceSelect.addEventListener("change", () => {

      if (voiceSelect.value === "cloud") {

        voiceStatusBadge.className = "badge cloud";

        voiceStatusBadge.textContent = "Cloud Neural Ready";

        modeHint.textContent = "Cloud Neural Active";

      } else {

        voiceStatusBadge.className = "badge ok";

        voiceStatusBadge.textContent = "Native Neural Ready";

        modeHint.textContent = "Device Neural Active";

      }

    });

    speakBtn.addEventListener("click", speakText);

    stopBtn.addEventListener("click", () => {

      stopAllAudio();

      statusText.textContent = "Stopped.";

    });

    downloadBtn.addEventListener("click", downloadAudio);

    if ("speechSynthesis" in window) {

      window.speechSynthesis.onvoiceschanged = loadSystemVoices;

    }

    selectProfile(PROFILES[0]);

    loadSystemVoices();

    setTimeout(loadSystemVoices, 400);
